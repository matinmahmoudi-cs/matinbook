\section{الگوریتم پایه}
\label{sec:basic-algorithm}

\begin{latin}
\begin{algorithm}
\caption{محاسبه فاکتوریل}
\label{alg:factorial}
\begin{algorithmic}[1]
    \Require عدد صحیح $n \geq 0$
    \Ensure $n!$
    \State $result \gets 1$
    \For{$i \gets 2$ \textbf{to} $n$}
        \State $result \gets result \times i$
    \EndFor
    \State \Return $result$
\end{algorithmic}
\end{algorithm}
\end{latin}

\section{الگوریتم جستجوی دودویی}
\label{sec:binary-search}

\begin{latin}
\begin{algorithm}
\caption{جستجوی دودویی}
\label{alg:binary-search}
\begin{algorithmic}[1]
    \Require آرایه مرتب $A$، مقدار $x$
    \Ensure اندیس $x$ یا $-1$
    \State $low \gets 0$
    \State $high \gets length(A) - 1$
    \While{$low \leq high$}
        \State $mid \gets \lfloor (low + high) / 2 \rfloor$
        \If{$A[mid] = x$}
            \State \Return $mid$
        \ElsIf{$A[mid] < x$}
            \State $low \gets mid + 1$
        \Else
            \State $high \gets mid - 1$
        \EndIf
    \EndWhile
    \State \Return $-1$
\end{algorithmic}
\end{algorithm}
\end{latin}

\section{الگوریتم مرتب‌سازی}
\label{sec:sorting}

\begin{latin}
\begin{algorithm}
\caption{مرتب‌سازی حبابی}
\label{alg:bubble-sort}
\begin{algorithmic}[1]
    \Require آرایه $A$ با $n$ عنصر
    \Ensure آرایه مرتب $A$
    \For{$i \gets 1$ \textbf{to} $n - 1$}
        \For{$j \gets 1$ \textbf{to} $n - i$}
            \If{$A[j] > A[j+1]$}
                \State \textbf{swap} $A[j]$ و $A[j+1]$
            \EndIf
        \EndFor
    \EndFor
\end{algorithmic}
\end{algorithm}
\end{latin}
