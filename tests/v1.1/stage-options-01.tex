\section{گزینه‌های فونت}
\label{sec:font-options}

گزینه‌های فونت حالا شامل موارد زیر هستند:

\begin{itemize}
    \item \verb|bnazanin| — B Nazanin (پیش‌فرض)
    \item \verb|vazirmatn| — Vazirmatn
    \item \verb|sahel| — Sahel
    \item \verb|irlotus| — IR Lotus
    \item \verb|niloofar| — XB Niloofar
\end{itemize}

\textbf{فونت پیش‌فرض:} B Nazanin

\textbf{بررسی:} اگر این متن با فونت B Nazanin نمایش داده شود،
سیستم گزینه‌ها موفق بوده است.

\section{محیط‌های علمی}
\label{sec:environments}

برای اطمینان از بارگذاری درست ماژول‌ها:

\begin{theorem}[قضیه نمونه]
    برای هر عدد حقیقی $x$، داریم $x^2 \geq 0$.
\end{theorem}

\begin{proof}
    چون مربع هر عدد حقیقی نامنفی است، نتیجه حاصل می‌شود.
\end{proof}

\begin{definition}[عدد اول]
    عدد طبیعی بزرگ‌تر از ۱ که تنها دو مقسوم‌علیه مثبت دارد،
    \emph{عدد اول} نامیده می‌شود.
\end{definition}
