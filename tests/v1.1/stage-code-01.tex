\section{کد پایه}
\label{sec:basic-code}

\begin{latin}
\begin{minted}{python}
def factorial(n):
    """Compute n! recursively."""
    if n <= 1:
        return 1
    return n * factorial(n - 1)

# Example usage
print(factorial(5))  # 120
\end{minted}
\end{latin}

function fibonacci(n) {
    if (n <= 1) return n;
    return fibonacci(n - 1) + fibonacci(n - 2);
}

console.log(fibonacci(10));  // 55
\end{minted}
\end{latin}
